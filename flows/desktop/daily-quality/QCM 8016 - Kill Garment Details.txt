
FUNCTION 'QCM 8016 - Kill Garment Details' GLOBAL
    IF (WebAutomation.IfWebPageContains.WebPageContainsElement BrowserInstance: Browser Control: appmask['Web Page \'h ... entDetails\'']['Div \'/QCM8016 - Kill Garment Details\'']) THEN
        SET SubflowRetryCount TO 0
        SET MaxSubflowRetries TO 2
        SET DownloadSuccess TO $'''False'''
        LOOP WHILE SubflowRetryCount < MaxSubflowRetries AND DownloadSuccess = $'''False'''
            WebAutomation.GoToWebPage.GoToWebPage BrowserInstance: Browser Url: Data8016 WaitForPageToLoadTimeout: 60
            WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() {
    return new Promise(function (resolve) {
        var targets = [
            \'[id*=\"chkbByDateRange_8016\"]\',
            \'[id*=\"chkbDefectInfo_8016\"]\',
            \'[id*=\"chkbShowGrades_8016\"]\'
        ];
        var results = [];
        var i = 0;

        function next() {
            if (i >= targets.length) { resolve(results.join(\" | \")); return; }
            var selector = targets[i];
            var attempts = 0, maxAttempts = 20, delayMs = 200; // ~4s ceiling

            (function poll() {
                var input = document.querySelector(selector);
                var wrapper = input && input.closest(\'.e-checkbox-wrapper\');
                var checkSpan = wrapper && wrapper.querySelector(\'span.e-frame\');

                if (checkSpan) {
                    if (!checkSpan.classList.contains(\'e-check\')) {
                        checkSpan.click();
                        results.push(selector + \": Clicked\");
                    } else {
                        results.push(selector + \": Already checked\");
                    }
                    i++; next();
                } else if (attempts++ < maxAttempts) {
                    setTimeout(poll, delayMs);
                } else {
                    results.push(selector + \": Not found after retries\");
                    i++; next();
                }
            })();
        }
        next();
    });
}'''
            ON ERROR REPEAT 5 TIMES WAIT 3
            END
            WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['Web Page \'h ... entDetails\'']['Input text \'DateRange_8016\'']) FOR 30
            ON ERROR WaitForWebPageContentError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            ON ERROR TimeoutError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            END
            @@RepairWithAIVision: 'True'
            WebAutomation.PopulateTextField.PopulateTextFieldUsePhysicalKeyboard BrowserInstance: Browser Control: appmask['Web Page \'h ... entDetails\'']['Input text \'DateRange_8016\''] Text: $'''%FormattedDateTime% - %FormattedDateTime%''' Mode: WebAutomation.PopulateTextMode.Replace UnfocusAfterPopulate: True WaitForPageToLoadTimeout: 60
            ON ERROR REPEAT 10 TIMES WAIT 3
            END
            LABEL 'Re-try for QCM 8016'
            WebAutomation.PressButton.PressButtonPressDialogButton BrowserInstance: Browser Control: appmask['Web Page \'h ... entDetails\'']['Button \'DOWNLOAD EXCEL\''] WaitForPageToLoadTimeout: 60 DialogButton: $'''OK'''
            ON ERROR PressButtonError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            END
            WAIT (UIAutomation.WaitForWindowContent.WindowToContainElement Element: appmask['Window \'Kill Garment Details - Google Chrome\'']['Window \'Save As\'']) FOR 300 
            ON ERROR WaitForWindowContentError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            ON ERROR TimeoutError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            END
            UIAutomation.PopulateTextField.PopulateTextField TextField: appmask['Window \'Kill Garment Details - Google Chrome\'']['Edit \'File name:\''] Text: $'''%Download_Path%\\%CurrentFactory%-%FormattedDateTimeForNaming%-Kill Garment Details for 8016.xls''' Mode: UIAutomation.PopulateTextMode.Replace ClickType: UIAutomation.PopulateMouseClickType.SingleClick
            UIAutomation.PressButton Button: appmask['Window \'Kill Garment Details - Google Chrome\'']['Button \'Save\''] Simulate: False
            ON ERROR PressButtonError
                SET SubflowRetryCount TO SubflowRetryCount + 1
                GOTO 'Retry QCM 8016'
            END
            WAIT (File.WaitForFile.Created File: $'''%Download_Path%\\%CurrentFactory%-%FormattedDateTimeForNaming%-Kill Garment Details for 8016.xls''') FOR 120 
            ON ERROR TimeoutError
                GOTO 'Re-try for QCM 8016'
            END
            SET DownloadSuccess TO $'''True'''
            LABEL 'Retry QCM 8016'
            IF DownloadSuccess = $'''False''' THEN
                WAIT 3
            END
        END
        IF (File.IfFile.DoesNotExist File: $'''%Download_Path%\\%CurrentFactory%-%FormattedDateTimeForNaming%-Kill Garment Details for 8016.xls''') THEN
            Variables.AddItemToList Item: $'''❌ %CurrentFactory%-%FormattedDateTimeForNaming%-Kill Garment Details for 8016 - Failed after %MaxSubflowRetries% attempts''' List: Missing_Files
        END
    ELSE
        Variables.AddItemToList Item: $'''⚠️ %CurrentFactory% - QCM 8016 - Report page not available''' List: Missing_Files
    END
END FUNCTION
