                resolve(\"Error: Input not found after retries.\");

            // Use Syncfusion\'s own API to open the popup
            if (input.ej2_instances && input.ej2_instances.length > 0) {

            if (wrapper) {
                wrapper.dispatchEvent(new MouseEvent(\'mousedown\', { bubbles: true }));
                wrapper.dispatchEvent(new MouseEvent(\'mouseup\',  { bubbles: true }));
                wrapper.dispatchEvent(new MouseEvent(\'click\',    { bubbles: true }));

            var input = document.querySelector(\'#msCheckpoint_2001\');
            var instance = (input && input.ej2_instances && input.ej2_instances.length > 0)
                ? input.ej2_instances[0] : null;

                targetValues.every(function (v) { return currentValues.indexOf(v) !== -1; });

            if (sameSet) {
                resolve(\"Skipped: Already selected.\");
                return;
            }

            instance.value = [];
            instance.value = targetValues;
