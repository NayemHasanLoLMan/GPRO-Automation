        var targets = [
            \'[id*=\"chkbDate_8012\"]\'
        ];
        var results = [];
        var i = 0;

        function next() {
            if (i >= targets.length) { resolve(results.join(\" | \")); return; }
            var selector = targets[i];
            var attempts = 0, maxAttempts = 20, delayMs = 200; // ~4s ceiling

            (function poll() {
                var input = document.querySelector(selector);
                var wrapper = input && input.closest(\'.e-checkbox-wrapper\');
                var checkSpan = wrapper && wrapper.querySelector(\'span.e-frame\');

                if (checkSpan) {
                    if (!checkSpan.classList.contains(\'e-check\')) {
                        checkSpan.click();
                        results.push(selector + \": Clicked\");
                    } else {
                        results.push(selector + \": Already checked\");
                    }
                    i++; next();
                } else if (attempts++ < maxAttempts) {
                    setTimeout(poll, delayMs);
                } else {
                    results.push(selector + \": Not found after retries\");
                    i++; next();
                }
            })();
        }
        next();
    });
}'''
