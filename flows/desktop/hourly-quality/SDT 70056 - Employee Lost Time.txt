
FUNCTION 'SDT 70056 - Employee Lost Time' GLOBAL

            SET SubflowRetryCount TO 0
            SET MaxSubflowRetries TO 2
            SET DownloadSuccess TO $'''False'''
            LOOP WHILE SubflowRetryCount < MaxSubflowRetries AND DownloadSuccess = $'''False'''
                WebAutomation.GoToWebPage.GoToWebPage BrowserInstance: Browser Url: Data70056 WaitForPageToLoadTimeout: 60
                WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['Web Page \'h ... eeLostTime\'']['Div \'Shift General\'']) FOR 30

                    GOTO 'Retry Dept 70056'
                ON ERROR TimeoutError
                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'
                END
                @@RepairWithAIVision: 'True'
WebAutomation.PopulateTextField.PopulateTextFieldUsePhysicalKeyboard BrowserInstance: Browser Control: appmask['Web Page \'h ... eeLostTime\'']['Input text \'combobox\' 2'] Text: Shift Mode: WebAutomation.PopulateTextMode.Replace UnfocusAfterPopulate: True WaitForPageToLoadTimeout: 60

                    GOTO 'Retry Dept 70056'
                ON ERROR TimeoutError
                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'
                END

                END

                ON ERROR WaitForWebPageContentError
                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'
                ON ERROR TimeoutError
                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'
                END
                @@RepairWithAIVision: 'True'

                ON ERROR REPEAT 10 TIMES WAIT 3
                END

                ON ERROR PressButtonError

                    GOTO 'Retry Dept 70056'
                END
                WAIT (UIAutomation.WaitForWindowContent.WindowToContainElement Element: appmask['Window \'70056 - Employee Lost Time - Google Chrome\'']['Window \'Save As\'']) FOR 300

                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'

                    SET SubflowRetryCount TO SubflowRetryCount + 1
                    GOTO 'Retry Dept 70056'

                    GOTO 'Retry Dept 70056'

                    GOTO 'Retry Download for SDT 70056'
                END
                SET DownloadSuccess TO $'''True'''
                LABEL 'Retry Dept 70056'
                IF DownloadSuccess = $'''False''' THEN
                    WAIT 3
                END
            END
            IF (File.IfFile.DoesNotExist File: $'''%Download_Path%\\%CurrentFactory%-%FormattedDateTimeForNaming%-Employee Lost Timer for 70056 (%CurrentDepertmentStage%).xls''') THEN
                Variables.AddItemToList Item: $'''❌ %CurrentFactory%-%FormattedDateTimeForNaming%-Employee Lost Timer for 70056 (%CurrentDepertmentStage%) - Failed after %MaxSubflowRetries% attempts''' List: Missing_Files

    END
END FUNCTION
