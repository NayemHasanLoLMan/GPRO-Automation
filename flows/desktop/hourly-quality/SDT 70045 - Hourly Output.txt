        var attempts = 0, maxAttempts = 20, delayMs = 200; // ~4s ceiling

        (function poll() {
            var input = document.querySelector(\'#msCheckpoint_70045\');

            if (!input) {
                if (attempts++ < maxAttempts) { setTimeout(poll, delayMs); return; }
                resolve(\"Error: Input not found after retries.\");
                return;
            }

            // Use Syncfusion\'s own API to open the popup
            if (input.ej2_instances && input.ej2_instances.length > 0) {
                input.ej2_instances[0].showPopup();
                resolve(\"Success: Opened via Syncfusion API.\");
                return;
            }

            var wrapper = input.closest(\'.e-multi-select-wrapper\');
            if (wrapper) {
                wrapper.dispatchEvent(new MouseEvent(\'mousedown\', { bubbles: true }));
                wrapper.dispatchEvent(new MouseEvent(\'mouseup\',  { bubbles: true }));
                wrapper.dispatchEvent(new MouseEvent(\'click\',    { bubbles: true }));
                resolve(\"Success: Opened via mouse events.\");
                return;

            resolve(\"Error: Could not open dropdown.\");
        })();
    });
}'''

        WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() {
    return new Promise(function (resolve) {
        var targetValues = [\"SEWING ENDLINE QC\", \"FIN OUT\"];
        var attempts = 0, maxAttempts = 20, delayMs = 200; // ~4s ceiling

        (function poll() {
            var input = document.querySelector(\'#msCheckpoint_70045\');
            var instance = (input && input.ej2_instances && input.ej2_instances.length > 0)
                ? input.ej2_instances[0] : null;

            if (!instance) {
                if (attempts++ < maxAttempts) { setTimeout(poll, delayMs); return; }
                resolve(input ? \"Error: Syncfusion instance not found after retries.\" : \"Error: Input not found after retries.\");
                return;
            }

            var currentValues = instance.value ? Array.from(instance.value) : [];
            var sameSet = currentValues.length === targetValues.length &&
                targetValues.every(function (v) { return currentValues.indexOf(v) !== -1; });

            if (sameSet) {
                resolve(\"Skipped: Already selected.\");
                return;
            }

            instance.value = [];
            instance.value = targetValues;

            resolve(\"Success: \" + instance.value.length + \" selected (\" + instance.value.join(\", \") + \").\");
        })();
    });
}'''
        ON ERROR REPEAT 5 TIMES WAIT 3
        END

    return new Promise(function (resolve) {
        var targets = [
            \'[id*=\"chkbIncludeBreakTime\"]\',
            \'[id*=\"chkbShowTargetDetails\"]\',
            \'[id*=\"chkbShowOrderDetails\"]\'
        ];
        var results = [];
        var i = 0;

        function next() {
            if (i >= targets.length) { resolve(results.join(\" | \")); return; }
            var selector = targets[i];
            var attempts = 0, maxAttempts = 20, delayMs = 200; // ~4s ceiling

            (function poll() {
                var input = document.querySelector(selector);
                var wrapper = input && input.closest(\'.e-checkbox-wrapper\');
                var checkSpan = wrapper && wrapper.querySelector(\'span.e-frame\');

                if (checkSpan) {
                    if (!checkSpan.classList.contains(\'e-check\')) {
                        checkSpan.click();
                        results.push(selector + \": Clicked\");
                    } else {
                        results.push(selector + \": Already checked\");
                    }
                    i++; next();
                } else if (attempts++ < maxAttempts) {
                    setTimeout(poll, delayMs);
                } else {
                    results.push(selector + \": Not found after retries\");
                    i++; next();
                }
            })();
        }
        next();
    });
}'''
